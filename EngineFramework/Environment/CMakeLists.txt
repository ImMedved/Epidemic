add_library(EpidemicGameFrameworkEnvironment STATIC src/environment.cpp)
target_include_directories(EpidemicGameFrameworkEnvironment PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(EpidemicGameFrameworkEnvironment PUBLIC EpidemicGameFrameworkFoundation PRIVATE EpidemicFoundation)
epidemic_configure_target(EpidemicGameFrameworkEnvironment)
set(_forbidden EpidemicGameFrameworkWorld EpidemicGameFrameworkInteraction EpidemicGameFrameworkRuntimeBridge EpidemicRuntimeEnvironment EpidemicRuntimePhysics EpidemicRuntimeWorld)
get_target_property(_links EpidemicGameFrameworkEnvironment LINK_LIBRARIES)
foreach(_t IN LISTS _forbidden)
  list(FIND _links "${_t}" _i)
  if(NOT _i EQUAL -1)
    message(FATAL_ERROR "GameFramework Environment must not link ${_t}")
  endif()
endforeach()
