add_library(EpidemicGameFrameworkInteraction STATIC src/interaction.cpp)
target_include_directories(EpidemicGameFrameworkInteraction PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(EpidemicGameFrameworkInteraction PUBLIC EpidemicGameFrameworkFoundation PRIVATE EpidemicFoundation)
epidemic_configure_target(EpidemicGameFrameworkInteraction)
set(_forbidden EpidemicGameFrameworkWorld EpidemicGameFrameworkEnvironment EpidemicGameFrameworkRuntimeBridge EpidemicRuntimePhysics EpidemicRuntimeWorld EpidemicGameplayQueries EpidemicGameplayFacts)
get_target_property(_links EpidemicGameFrameworkInteraction LINK_LIBRARIES)
foreach(_t IN LISTS _forbidden)
  list(FIND _links "${_t}" _i)
  if(NOT _i EQUAL -1)
    message(FATAL_ERROR "GameFramework Interaction must not link ${_t}")
  endif()
endforeach()
