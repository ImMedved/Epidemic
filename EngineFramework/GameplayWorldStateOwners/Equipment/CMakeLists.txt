add_library(EpidemicGameFrameworkEquipment STATIC src/equipment.cpp)
target_include_directories(EpidemicGameFrameworkEquipment PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(EpidemicGameFrameworkEquipment PUBLIC EpidemicGameFrameworkFoundation PRIVATE EpidemicFoundation)
epidemic_configure_target(EpidemicGameFrameworkEquipment)
set(_forbidden EpidemicGameFrameworkItemsInventory EpidemicGameFrameworkProgression EpidemicGameFrameworkCombat EpidemicGameFrameworkAbilities)
get_target_property(_links EpidemicGameFrameworkEquipment LINK_LIBRARIES)
if(_links STREQUAL "_links-NOTFOUND")
 set(_links "")
endif()
foreach(_t IN LISTS _forbidden)
 list(FIND _links "${_t}" _idx)
 if(NOT _idx EQUAL -1)
  message(FATAL_ERROR "Equipment core must not link peer major '${_t}'.")
 endif()
endforeach()
