add_library(EpidemicGameFrameworkWorld STATIC src/world.cpp)
target_include_directories(EpidemicGameFrameworkWorld PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(EpidemicGameFrameworkWorld PUBLIC EpidemicGameFrameworkFoundation PRIVATE EpidemicFoundation)
epidemic_configure_target(EpidemicGameFrameworkWorld)
set(_forbidden EpidemicGameFrameworkRuntimeBridge EpidemicRuntimeWorld EpidemicRuntimePhysics EpidemicRuntimeEnvironment EpidemicRuntimeStreaming EpidemicGameFrameworkEnvironment EpidemicGameFrameworkInteraction)
get_target_property(_links EpidemicGameFrameworkWorld LINK_LIBRARIES)
foreach(_t IN LISTS _forbidden)
  list(FIND _links "${_t}" _i)
  if(NOT _i EQUAL -1)
    message(FATAL_ERROR "GameFramework World must not link ${_t}")
  endif()
endforeach()
