add_library(EpidemicGameFrameworkDialogue STATIC src/dialogue.cpp)
target_include_directories(EpidemicGameFrameworkDialogue PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(EpidemicGameFrameworkDialogue PUBLIC EpidemicGameFrameworkFoundation PRIVATE EpidemicFoundation)
epidemic_configure_target(EpidemicGameFrameworkDialogue)
set(_forbidden EpidemicGameFrameworkNarrative EpidemicGameFrameworkKnowledge EpidemicGameFrameworkSociety EpidemicGameFrameworkCrime EpidemicGameFrameworkEconomy EpidemicGameFrameworkAI)
get_target_property(_links EpidemicGameFrameworkDialogue LINK_LIBRARIES)
if(_links STREQUAL "_links-NOTFOUND")
 set(_links "")
endif()
foreach(_t IN LISTS _forbidden)
 list(FIND _links "${_t}" _idx)
 if(NOT _idx EQUAL -1)
  message(FATAL_ERROR "Dialogue core must not link peer major '${_t}'.")
 endif()
endforeach()
