add_library(EpidemicGameFrameworkItemsInventory STATIC src/items_inventory.cpp)
target_include_directories(EpidemicGameFrameworkItemsInventory PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(EpidemicGameFrameworkItemsInventory PUBLIC EpidemicGameFrameworkFoundation PRIVATE EpidemicFoundation)
epidemic_configure_target(EpidemicGameFrameworkItemsInventory)
set(_epidemic_items_forbidden_targets EpidemicGameFrameworkEquipment EpidemicGameFrameworkEconomy EpidemicGameFrameworkOwnership EpidemicGameFrameworkLoot EpidemicGameFrameworkProcesses)
get_target_property(_epidemic_items_links EpidemicGameFrameworkItemsInventory LINK_LIBRARIES)
if(_epidemic_items_links STREQUAL "_epidemic_items_links-NOTFOUND")
  set(_epidemic_items_links "")
endif()
foreach(_forbidden_target IN LISTS _epidemic_items_forbidden_targets)
  list(FIND _epidemic_items_links "${_forbidden_target}" _idx)
  if(NOT _idx EQUAL -1)
    message(FATAL_ERROR "ItemsInventory core must not link peer major '${_forbidden_target}'.")
  endif()
endforeach()
