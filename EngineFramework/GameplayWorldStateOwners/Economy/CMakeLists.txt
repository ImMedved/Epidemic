add_library(EpidemicGameFrameworkEconomy STATIC src/economy.cpp)
target_include_directories(EpidemicGameFrameworkEconomy PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(EpidemicGameFrameworkEconomy PUBLIC EpidemicGameFrameworkFoundation PRIVATE EpidemicFoundation)
epidemic_configure_target(EpidemicGameFrameworkEconomy)
set(_forbidden EpidemicGameFrameworkItemsInventory EpidemicGameFrameworkOwnership EpidemicGameFrameworkResourcesProduction EpidemicGameFrameworkSociety EpidemicGameFrameworkRolesJobs)
get_target_property(_links EpidemicGameFrameworkEconomy LINK_LIBRARIES)
if(_links STREQUAL "_links-NOTFOUND")
 set(_links "")
endif()
foreach(_t IN LISTS _forbidden)
 list(FIND _links "${_t}" _idx)
 if(NOT _idx EQUAL -1)
  message(FATAL_ERROR "Economy core must not link peer major '${_t}'.")
 endif()
endforeach()
